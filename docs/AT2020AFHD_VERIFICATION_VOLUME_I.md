def lomb_scargle_analysis(time_mjd, flux, flux_err):
    """Calcula periodograma Lomb-Scargle"""
    # Convertir a días desde época
    time_days = time_mjd - time_mjd[0]
    
    # Rango de frecuencias (1/días)
    freq_min = 0.001
    freq_max = 0.1
    freq = np.linspace(freq_min, freq_max, 10000)
    
    # Calcular periodograma
    ls = LombScargle(time_days, flux, flux_err)
    power = ls.power(freq)
    
    # Encontrar pico máximo
    idx_max = np.argmax(power)
    freq_peak = freq[idx_max]
    period_peak = 1.0 / freq_peak
    
    return freq, power, period_peak

def calculate_harmonic_ratio(f0_hz, period_days):
    """Calcula ratio armónico y octavas"""
    # Convertir periodo a frecuencia en Hz
    f_frame = 1.0 / (period_days * 86400)  # 86400 s/día
    
    # Ratio armónico
    ratio = f0_hz / f_frame
    
    # Octavas
    octaves = np.log2(ratio)
    
    # Órdenes de magnitud
    orders = np.log10(ratio)
    
    return {
        'f_frame': f_frame,
        'ratio': ratio,
        'octaves': octaves,
        'orders': orders
    }
